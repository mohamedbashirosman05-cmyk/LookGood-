  function show(list){
    res.textContent="";

    var q=inp.value.trim(),my=++seq;clearTimeout(timer);
    if(!q){res.textContent="";return}
    var L=local(q);show(L);
    timer=setTimeout(function(){
      online(q).then(function(O){if(my!==seq)return;
        var out=O.slice();L.forEach(function(c){if(!out.some(function(o){return near(o,c)}))out.push(c)});

        if(out.length)show(out);else msg("Keine Stadt gefunden. Prüfe die Schreibweise.")})
      .catch(function(){if(my===seq&&!L.length)msg("Keine Treffer in der Liste. Die Online-Suche ist in dieser Vorschau nicht erreichbar.")})},250)}

  inp.onkeydown=function(e){if(e.key==="Enter"){var f=res.querySelector("button");if(f)f.click()}};
